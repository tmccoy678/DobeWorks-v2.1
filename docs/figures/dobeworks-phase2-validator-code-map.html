0231  def collect_bounded_output(
0232      process: subprocess.Popen[bytes],
0233      label: str,
0234      timeout_seconds: int,
0235  ) -> tuple[bytes, bytes]:
0236      selector = selectors.DefaultSelector()
0237      buffers = {"stdout": bytearray(), "stderr": bytearray()}
0238      streams = {"stdout": process.stdout, "stderr": process.stderr}
0239      deadline = time.monotonic() + timeout_seconds
0240      try:
0241          for name, stream in streams.items():
0242              if stream is not None:
0243                  selector.register(stream, selectors.EVENT_READ, name)
0244          while selector.get_map():
0245              remaining = deadline - time.monotonic()
0246              events = selector.select(max(0, remaining))
0247              if remaining <= 0 or not events:
0248                  terminate_process_group(process)
0249                  raise ValidationError(f"{label} read timed out")
0250              for key, _ in events:
0251                  used = sum(len(item) for item in buffers.values())
0252                  capacity = MAX_TEXT_BYTES - used
0253                  chunk = os.read(key.fileobj.fileno(), min(65536, capacity + 1))
0254                  if not chunk:
0255                      selector.unregister(key.fileobj)
0256                  elif len(chunk) > capacity:
0257                      terminate_process_group(process)
0258                      raise ValidationError(f"{label} output limit exceeded")
0259                  else:
0260                      buffers[key.data].extend(chunk)
0261          process.wait(timeout=max(0, deadline - time.monotonic()))
0262      except subprocess.TimeoutExpired as error:
0263          terminate_process_group(process)
0264          raise ValidationError(f"{label} read timed out") from error
0265      finally:
0266          selector.close()
0267          for stream in streams.values():
0268              if stream is not None:
0269                  stream.close()
0270      return bytes(buffers["stdout"]), bytes(buffers["stderr"])

0293  def parse_manifest(
0294      root: Path,
0295      relative: str,
0296      expected_paths: frozenset[str],
0297  ) -> dict[str, str]:
0298      text = rooted_text(root, relative)
0299      entries: dict[str, str] = {}
0300      ordered: list[str] = []
0301      for line in text.splitlines():
0302          match = MANIFEST_LINE.fullmatch(line)
0303          if match is None:
0304              raise ValidationError(f"manifest contains a malformed line: {relative}")
0305          digest, artifact = match.groups()
0306          if artifact in entries:
0307              raise ValidationError(f"manifest contains a duplicate path: {relative}")
0308          entries[artifact] = digest
0309          ordered.append(artifact)
0310      if ordered != sorted(ordered) or frozenset(ordered) != expected_paths:
0311          raise ValidationError(f"manifest path set or order differs: {relative}")
0312      return entries

Phase 2 | the package must be the package

The 792-line DobeWorks validator is separate from the 1,213-line corrected DEGS evaluator.

Figure 1
Exact identity before package acceptance

1. Identify

Use the exact expected package surface, source pins, and externally supplied manifest digest.

2. Preserve

Check historical observations, generation records, lifecycle fields, and required evidence structure.

3. Verify

Run the pinned evaluator through bounded, checked interfaces and require exact accepted output.

4. Report

Return package PASS only if every required check agrees. Keep review, delivery, qualification and release separate.

Why this core is worth preserving

Its value is the chain of explicit checks: controlled inputs, bounded operations, preserved evidence, checked subprocess results, exact identities, and a result whose scope is visible. The design connects source discipline with systems-engineering traceability (Holzmann, 2006; Hirshorn et al., 2017).

Historical package PASS describes the frozen package and observation. It does not establish that a later staged checkout still matches those bytes. A documentation change can invalidate an exact manifest without changing the runtime implementation.
